# gRPC server. The request policy (limits, spec validation, preset headroom) is plain C++ and
# tested on the CPU. The service and the ombra_server executable need CUDA (the search engine) and
# gRPC for C++ (Ubuntu 22.04: libgrpc++-dev libprotobuf-dev protobuf-compiler
# protobuf-compiler-grpc); without either, only the policy is built and the CPU build is unchanged.

# ---- Preset table, generated from the preset JSON (no hand-picked numbers) ----------
set(OMBRA_PRESET_JSON ${CMAKE_CURRENT_SOURCE_DIR}/../../core/params/ckks_presets.json)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${OMBRA_PRESET_JSON})
file(READ ${OMBRA_PRESET_JSON} _presets_json)
string(JSON _count LENGTH ${_presets_json} presets)
math(EXPR _last "${_count} - 1")
set(_entries "")
foreach(_i RANGE ${_last})
  string(JSON _name MEMBER ${_presets_json} presets ${_i})
  string(JSON _sec GET ${_presets_json} presets ${_name} security_level)
  if(NOT _sec MATCHES "^HEStd_(128|192|256)_(classic|quantum)$")
    message(FATAL_ERROR "preset ${_name}: security level ${_sec} is below 128 bits")
  endif()
  string(JSON _ring GET ${_presets_json} presets ${_name} ring_dim)
  string(JSON _batch GET ${_presets_json} presets ${_name} batch_size)
  string(JSON _first GET ${_presets_json} presets ${_name} first_mod_size)
  string(JSON _scale GET ${_presets_json} presets ${_name} scaling_mod_size)
  string(APPEND _entries "    {\"${_name}\", ${_ring}, ${_batch}, ${_first}, ${_scale}},\n")
endforeach()
set(_gen_dir ${CMAKE_CURRENT_BINARY_DIR}/gen)
file(WRITE ${_gen_dir}/preset_table.hpp.tmp
"// Generated by CMake from core/params/ckks_presets.json. Do not edit.
#pragma once
#include \"ombra/server/policy.hpp\"
namespace ombra::server::generated {
inline constexpr PresetInfo kPresets[] = {
${_entries}};
}  // namespace ombra::server::generated
")
file(COPY_FILE ${_gen_dir}/preset_table.hpp.tmp ${_gen_dir}/preset_table.hpp ONLY_IF_DIFFERENT)

add_library(ombra_server_policy STATIC src/policy.cpp)
target_include_directories(ombra_server_policy PUBLIC include PRIVATE ${_gen_dir})
target_link_libraries(ombra_server_policy PUBLIC ombra_search_logic)

add_executable(server_policy_test tests/policy_test.cpp)
target_link_libraries(server_policy_test PRIVATE ombra_server_policy)
add_test(NAME server_policy COMMAND server_policy_test)

# ---- API keys, metrics, HTTP parsing: plain C++ (own SHA-256, no OpenSSL), CPU tests ---------
add_library(ombra_server_admin STATIC src/auth.cpp src/metrics.cpp src/http.cpp)
target_include_directories(ombra_server_admin PUBLIC include)

add_executable(server_admin_test tests/admin_test.cpp)
target_link_libraries(server_admin_test PRIVATE ombra_server_admin)
add_test(NAME server_admin COMMAND server_admin_test)

# ---- gRPC service and executable ----------------------------------------------------------------
if(NOT OMBRA_ENABLE_CUDA)
  return()
endif()

# Discovery: the CMake config of gRPC (or a plain pkg-config file), protobuf, protoc and the gRPC
# C++ plugin. All must be found, otherwise the server is skipped with a status message.
find_package(gRPC CONFIG QUIET)
set(_grpc_libs "")
if(gRPC_FOUND)
  set(_grpc_libs gRPC::grpc++)
else()
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(OMBRA_GRPCPP QUIET IMPORTED_TARGET grpc++)
    if(OMBRA_GRPCPP_FOUND)
      set(_grpc_libs PkgConfig::OMBRA_GRPCPP)
    endif()
  endif()
endif()
find_package(Protobuf QUIET)
find_program(OMBRA_PROTOC protoc)
find_program(OMBRA_GRPC_CPP_PLUGIN grpc_cpp_plugin)

if(NOT _grpc_libs OR NOT Protobuf_FOUND OR NOT OMBRA_PROTOC OR NOT OMBRA_GRPC_CPP_PLUGIN)
  message(STATUS "ombra_server not built: gRPC for C++ not found "
                 "(libgrpc++-dev libprotobuf-dev protobuf-compiler protobuf-compiler-grpc)")
  return()
endif()
message(STATUS "ombra_server: gRPC found (protoc ${OMBRA_PROTOC}, plugin ${OMBRA_GRPC_CPP_PLUGIN})")

set(_proto_root ${CMAKE_CURRENT_SOURCE_DIR}/../../proto)
set(_proto ${_proto_root}/ombra/v1/ombra.proto)
set(_pb_dir ${CMAKE_CURRENT_BINARY_DIR}/proto_gen)
file(MAKE_DIRECTORY ${_pb_dir})
set(_pb_srcs ${_pb_dir}/ombra/v1/ombra.pb.cc ${_pb_dir}/ombra/v1/ombra.grpc.pb.cc)
add_custom_command(
  OUTPUT ${_pb_srcs} ${_pb_dir}/ombra/v1/ombra.pb.h ${_pb_dir}/ombra/v1/ombra.grpc.pb.h
  COMMAND ${OMBRA_PROTOC} -I ${_proto_root} --cpp_out=${_pb_dir} --grpc_out=${_pb_dir}
          --plugin=protoc-gen-grpc=${OMBRA_GRPC_CPP_PLUGIN} ${_proto}
  DEPENDS ${_proto}
  COMMENT "Generating C++ gRPC code from ombra.proto")

find_package(Threads REQUIRED)
add_library(ombra_proto STATIC ${_pb_srcs})
target_include_directories(ombra_proto SYSTEM PUBLIC ${_pb_dir})
target_link_libraries(ombra_proto PUBLIC ${_grpc_libs} protobuf::libprotobuf Threads::Threads)

# cudaMemGetInfo for /metrics. CMake 3.22 has no CUDA20 dialect, so CUDA_STANDARD 17 as elsewhere.
add_library(ombra_gpu_info STATIC src/gpu_info.cu)
target_include_directories(ombra_gpu_info PUBLIC src)
set_target_properties(ombra_gpu_info PROPERTIES CUDA_SEPARABLE_COMPILATION OFF CUDA_STANDARD 17)

# The REST port speaks TLS with OpenSSL, which gRPC already depends on.
find_package(OpenSSL REQUIRED)

add_executable(ombra_server src/ombra_server.cpp src/service.cpp src/rest_server.cpp)
target_include_directories(ombra_server PRIVATE src)
target_link_libraries(ombra_server PRIVATE ombra_server_policy ombra_server_admin ombra_gpu_info
                                           ombra_search ombra_proto OpenSSL::SSL OpenSSL::Crypto)
target_compile_definitions(ombra_server PRIVATE OMBRA_VERSION="${PROJECT_VERSION}")
# Default OpenFHE version of the stored envelopes (overridable with --openfhe-version). The
# variable of core/'s find_package is directory-scoped, so look the package up again here.
find_package(OpenFHE QUIET)
if(DEFINED OpenFHE_VERSION)
  target_compile_definitions(ombra_server PRIVATE OMBRA_OPENFHE_VERSION="${OpenFHE_VERSION}")
endif()
